## 🌱 Start here - what "automation" actually means (and how to think about it)

Automation is just this: **you do a boring task once, by hand, slowly; then you teach the computer to do it a thousand times, instantly, while you get coffee.** That's the entire game. The famous book *Automate the Boring Stuff with Python* is built on this one idea, and it's the most directly *useful* skill in this whole textbook, it's the thing that makes non-programmers say "wait, you can do that?"

Every single project in this chapter, and almost every automation you'll ever write, is the same four-beat loop:

> **read input → transform → write output → (optionally) notify or schedule**

Once you see that shape, automations stop feeling like separate tricks and start feeling like the same recipe with different ingredients:
- *Photo organizer:* read photos → figure out each date → move into dated folders → done.
- *Mailer:* read a CSV → render a message per row → send each → log it.
- *Weather dashboard:* fetch a number → append to history → rebuild an HTML page → run hourly.

**How to find your own automation projects:** notice the moment you sigh and think "ugh, I have to do this *again*." Renaming files one by one. Copy-pasting data between a spreadsheet and an email. Checking a website every morning. Each sigh is a candidate. The best first automations are small, repetitive, and rule-based, exactly the things computers are great at and humans are bad at staying patient for.

**Two safety habits to burn in before we start**, because automation acts fast and at scale (a bug doesn't ruin one file, it ruins a thousand):

1. **Dry-run by default.** Any tool that moves, renames, deletes, or sends should first *print what it would do* and only act when you pass `--apply`. You'd rather read a plan than recover from a disaster.
2. **Never destroy the original.** Write to a *new* file, copy before transforming, keep a log of what changed. Reversibility is cheap insurance.

With that mindset locked in, let's build five complete tools, then four more "scratch a real itch" mini-automations after them.

## 🧪 Project 1 - Bulk photo renamer / EXIF organizer

> Goal: Take a folder of camera dumps (`IMG_4821.JPG`, `DSC0093.jpg`) and rename/organize them by the date the photo was *actually taken* (from EXIF metadata), into `YYYY/YYYY-MM-DD_HHMMSS.jpg`. Dry-run by default.

### Design
1. Walk the source folder for image files.
2. For each, read EXIF (`DateTimeOriginal`) via Pillow; fall back to the file's modified time if EXIF is missing.
3. Build a date-based destination path and a collision-safe filename.
4. Move (or copy), only when `--apply` is passed.

EXIF is metadata the camera embeds: timestamp, camera model, GPS, orientation. The tag you care about is `DateTimeOriginal` (tag id 36867).

### Full code

```python
#!/usr/bin/env python3
"""photo_organizer.py - rename/organize photos by EXIF capture date.

Usage:
    python photo_organizer.py ~/CameraDump                 # dry run
    python photo_organizer.py ~/CameraDump --apply         # move files
    python photo_organizer.py ~/CameraDump --copy --apply  # copy instead of move
"""
from __future__ import annotations

import argparse
import shutil
from datetime import datetime
from pathlib import Path

from PIL import Image
from PIL.ExifTags import TAGS

IMAGE_EXTS = {".jpg", ".jpeg", ".png", ".tiff", ".heic"}
EXIF_DATE_TAG = "DateTimeOriginal"      # EXIF tag id 36867


def capture_time(path: Path) -> datetime:
    """Return the photo's capture time from EXIF, or fall back to file mtime."""
    try:
        with Image.open(path) as img:
            exif = img.getexif()
            for tag_id, value in exif.items():
                if TAGS.get(tag_id) == EXIF_DATE_TAG:
                    # EXIF format: "2026:06:26 14:30:05"
                    return datetime.strptime(str(value), "%Y:%m:%d %H:%M:%S")
    except Exception:
        pass  # not an image we can read, or no EXIF -> fall through
    return datetime.fromtimestamp(path.stat().st_mtime)


def unique(dest: Path) -> Path:
    if not dest.exists():
        return dest
    n = 1
    while True:
        candidate = dest.with_stem(f"{dest.stem}_{n}")
        if not candidate.exists():
            return candidate
        n += 1


def organize(src: Path, apply: bool, copy: bool) -> None:
    if not src.is_dir():
        raise SystemExit(f"Not a directory: {src}")

    count = 0
    for item in sorted(src.iterdir()):
        if not item.is_file() or item.suffix.lower() not in IMAGE_EXTS:
            continue
        dt = capture_time(item)
        dest_dir = src / dt.strftime("%Y") / dt.strftime("%Y-%m")
        new_name = dt.strftime("%Y-%m-%d_%H%M%S") + item.suffix.lower()
        dest = unique(dest_dir / new_name)

        action = ("COPY" if copy else "MOVE") if apply else "PLAN"
        print(f"{action}  {item.name}  ->  {dest.relative_to(src)}")

        if apply:
            dest.parent.mkdir(parents=True, exist_ok=True)
            if copy:
                shutil.copy2(item, dest)     # copy2 preserves metadata
            else:
                shutil.move(str(item), str(dest))
        count += 1

    print(f"\n{'Processed' if apply else 'Would process'} {count} photo(s).")
    if not apply:
        print("Re-run with --apply to actually do it.")


def main() -> None:
    ap = argparse.ArgumentParser(description="Organize photos by EXIF date.")
    ap.add_argument("folder", type=Path)
    ap.add_argument("--apply", action="store_true", help="actually move/copy (default: dry run)")
    ap.add_argument("--copy", action="store_true", help="copy instead of move")
    args = ap.parse_args()
    organize(args.folder.expanduser(), apply=args.apply, copy=args.copy)


if __name__ == "__main__":
    main()
```


## 🧪 Project 2 - CSV-driven templated email/report sender

> Goal: Read a CSV of recipients + per-row data, render a personalized email from a template, and send each via SMTP. This mirrors real email-pipeline work, the core loop of "data row → templated message → deliver" is exactly what production email systems do, just bigger.

### Design
1. Load recipients from CSV (`name,email,balance,due_date`).
2. Render subject + body per row with a template (Jinja2, or `str.format` for simple cases).
3. Connect to an SMTP server over TLS, authenticated with env-var credentials.
4. Send; log successes/failures; support `--dry-run` to preview without sending.

**Critical safety:** credentials come from environment variables (Chapter 4.2), never the source. For Gmail, that means an *app password*, not the real account password.

### Full code

```python
#!/usr/bin/env python3
"""mailer.py - send personalized emails from a CSV using a template.

Setup (env vars; never hard-code):
    export SMTP_HOST=smtp.gmail.com
    export SMTP_PORT=587
    export SMTP_USER=you@gmail.com
    export SMTP_PASS=your_app_password

Usage:
    python mailer.py recipients.csv --dry-run     # render + preview, send nothing
    python mailer.py recipients.csv               # actually send
"""
from __future__ import annotations

import argparse
import csv
import os
import smtplib
import ssl
import sys
import time
from email.message import EmailMessage
from pathlib import Path

# Simple, dependency-free templating with str.format. (Swap in Jinja2 for loops/conditionals.)
SUBJECT_TMPL = "Account update for {name}"
BODY_TMPL = """\
Hi {name},

This is your monthly summary.

  Current balance : ${balance}
  Payment due by  : {due_date}

Thanks,
Spencer's Automated Mailer
"""


def load_recipients(path: Path) -> list[dict]:
    with path.open(newline="", encoding="utf-8") as f:
        return list(csv.DictReader(f))


def build_message(row: dict, sender: str) -> EmailMessage:
    msg = EmailMessage()
    msg["From"] = sender
    msg["To"] = row["email"]
    msg["Subject"] = SUBJECT_TMPL.format(**row)
    msg.set_content(BODY_TMPL.format(**row))
    return msg


def send_all(rows: list[dict], dry_run: bool) -> None:
    host = os.environ.get("SMTP_HOST", "smtp.gmail.com")
    port = int(os.environ.get("SMTP_PORT", "587"))
    user = os.environ.get("SMTP_USER")
    password = os.environ.get("SMTP_PASS")

    if not dry_run and (not user or not password):
        sys.exit("Set SMTP_USER and SMTP_PASS env vars (use an app password for Gmail).")

    sender = user or "noreply@example.com"
    sent, failed = 0, 0

    if dry_run:
        for row in rows:
            msg = build_message(row, sender)
            print("=" * 60)
            print(f"To: {msg['To']}\nSubject: {msg['Subject']}\n")
            print(msg.get_content())
        print(f"\nDRY RUN - would send {len(rows)} email(s). Nothing was sent.")
        return

    # Real send: one authenticated TLS connection, reused for all messages.
    context = ssl.create_default_context()
    with smtplib.SMTP(host, port) as server:
        server.starttls(context=context)        # upgrade the connection to encrypted TLS
        server.login(user, password)
        for row in rows:
            try:
                server.send_message(build_message(row, sender))
                print(f"sent -> {row['email']}")
                sent += 1
                time.sleep(0.5)                  # gentle throttle; avoid provider rate limits
            except Exception as e:
                print(f"FAILED -> {row['email']}: {e}")
                failed += 1

    print(f"\nDone. Sent {sent}, failed {failed}.")


def main() -> None:
    ap = argparse.ArgumentParser(description="Send templated emails from a CSV.")
    ap.add_argument("csv_path", type=Path)
    ap.add_argument("--dry-run", action="store_true", help="render + preview, send nothing")
    args = ap.parse_args()
    rows = load_recipients(args.csv_path)
    if not rows:
        sys.exit("No rows in CSV.")
    send_all(rows, dry_run=args.dry_run)


if __name__ == "__main__":
    main()
```

Example `recipients.csv`:

```csv
name,email,balance,due_date
Alice,alice@example.com,120.50,2026-07-15
Bob,bob@example.com,0.00,2026-07-15
```

### How to run
```bash
python mailer.py recipients.csv --dry-run     # ALWAYS preview first
export SMTP_USER=you@gmail.com SMTP_PASS=your_app_password
python mailer.py recipients.csv
```

### What ties to real email pipelines
Production senders add: a queue (so a crash doesn't lose messages), idempotency (don't double-send on retry), bounce/complaint handling, HTML+plaintext multipart bodies, and per-domain rate limiting. The loop here is the kernel of all of that, `--dry-run` is the "staging" safety you'd never ship without.

### Extend it
- Use **Jinja2** templates (`{% if balance %}...{% endif %}`) and HTML bodies via `msg.add_alternative(html, subtype="html")`.
- Attach a per-recipient PDF (see Project 5).
- Write a `sent.log` and skip anyone already in it (idempotency).

## 🧪 Project 3 - Spreadsheet automator with openpyxl

> Goal: Read a sales spreadsheet, compute per-category totals and a grand total, write a clean summary sheet, and embed a bar chart, all without opening Excel. `openpyxl` reads/writes `.xlsx` natively.

### Design
1. Read the `Sales` sheet (columns: `date, category, amount`).
2. Aggregate `amount` by `category`.
3. Write a `Summary` sheet with totals, formatting (bold header, currency format).
4. Add a `BarChart` referencing the summary data.
5. Save to a new file (never overwrite the source).

### Full code

```python
#!/usr/bin/env python3
"""sales_report.py - summarize sales.xlsx into a formatted report with a chart.

Usage:
    python sales_report.py sales.xlsx report.xlsx
"""
from __future__ import annotations

import sys
from collections import defaultdict
from pathlib import Path

from openpyxl import Workbook, load_workbook
from openpyxl.chart import BarChart, Reference
from openpyxl.styles import Font, PatternFill


def read_sales(path: Path) -> dict[str, float]:
    wb = load_workbook(path, read_only=True, data_only=True)   # data_only -> values, not formulas
    ws = wb.active
    totals: dict[str, float] = defaultdict(float)
    rows = ws.iter_rows(min_row=2, values_only=True)           # skip header row
    for date, category, amount in rows:
        if category is None or amount is None:
            continue
        totals[str(category)] += float(amount)
    wb.close()
    return dict(totals)


def write_report(totals: dict[str, float], out: Path) -> None:
    wb = Workbook()
    ws = wb.active
    ws.title = "Summary"

    # Header row with styling
    headers = ["Category", "Total"]
    ws.append(headers)
    bold = Font(bold=True, color="FFFFFF")
    fill = PatternFill("solid", fgColor="305496")
    for cell in ws[1]:
        cell.font = bold
        cell.fill = fill

    # Data rows, sorted by total descending
    for category, total in sorted(totals.items(), key=lambda kv: kv[1], reverse=True):
        ws.append([category, total])

    # Grand total
    grand = sum(totals.values())
    ws.append(["GRAND TOTAL", grand])
    last_row = ws.max_row
    ws.cell(row=last_row, column=1).font = Font(bold=True)
    ws.cell(row=last_row, column=2).font = Font(bold=True)

    # Currency format for the Total column
    for row in ws.iter_rows(min_row=2, min_col=2, max_col=2):
        for cell in row:
            cell.number_format = '"$"#,##0.00'

    ws.column_dimensions["A"].width = 22
    ws.column_dimensions["B"].width = 16

    # Bar chart of category totals (exclude the grand-total row)
    chart = BarChart()
    chart.title = "Sales by Category"
    chart.y_axis.title = "Total ($)"
    chart.x_axis.title = "Category"
    data = Reference(ws, min_col=2, min_row=1, max_row=last_row - 1)   # include header for legend
    cats = Reference(ws, min_col=1, min_row=2, max_row=last_row - 1)
    chart.add_data(data, titles_from_data=True)
    chart.set_categories(cats)
    ws.add_chart(chart, "D2")

    wb.save(out)


def main() -> None:
    if len(sys.argv) != 3:
        sys.exit("Usage: python sales_report.py <in.xlsx> <out.xlsx>")
    src, out = Path(sys.argv[1]), Path(sys.argv[2])
    totals = read_sales(src)
    write_report(totals, out)
    print(f"Wrote {out} with {len(totals)} categories.")
    for cat, total in sorted(totals.items(), key=lambda kv: -kv[1]):
        print(f"  {cat:<16} ${total:,.2f}")


if __name__ == "__main__":
    main()
```

## 🧪 Project 4 - Scheduled scraper → dashboard pipeline

> Goal: A small recurring pipeline: fetch data from a free API (Open-Meteo weather, no key needed), append a timestamped record to a history file, and regenerate a self-contained HTML dashboard. Then schedule it to run automatically (cron). This is the "read → transform → write → publish" loop on a timer.

### Design
1. **Fetch:** call the Open-Meteo API for a city's current temperature.
2. **Store:** append `{timestamp, temp}` to `history.json` (the accumulating dataset).
3. **Render:** write `dashboard.html` with the latest reading and a tiny inline SVG sparkline of recent history, no JS framework, no build step.
4. **Schedule:** a cron line runs the script every hour.

### Full code

```python
#!/usr/bin/env python3
"""weather_dashboard.py - fetch temp, append history, regenerate an HTML dashboard.

Usage:
    python weather_dashboard.py            # one run: fetch + update history + rebuild dashboard
"""
from __future__ import annotations

import json
from datetime import datetime
from pathlib import Path

import requests

LAT, LON, CITY = 40.76, -111.89, "Salt Lake City"   # change to your location
HISTORY = Path("history.json")
DASHBOARD = Path("dashboard.html")
API = "https://api.open-meteo.com/v1/forecast"


def fetch_temp() -> float:
    r = requests.get(API, params={"latitude": LAT, "longitude": LON, "current_weather": True},
                     timeout=10)
    r.raise_for_status()
    return float(r.json()["current_weather"]["temperature"])   # Celsius


def load_history() -> list[dict]:
    if HISTORY.exists():
        return json.loads(HISTORY.read_text(encoding="utf-8"))
    return []


def save_history(history: list[dict]) -> None:
    HISTORY.write_text(json.dumps(history, indent=2), encoding="utf-8")


def sparkline_svg(temps: list[float], width: int = 600, height: int = 120) -> str:
    """Hand-built SVG polyline - no charting library needed."""
    if len(temps) < 2:
        return "<p>Not enough data yet for a chart.</p>"
    lo, hi = min(temps), max(temps)
    span = (hi - lo) or 1
    n = len(temps)
    points = []
    for i, t in enumerate(temps):
        x = i / (n - 1) * (width - 20) + 10
        y = height - 10 - (t - lo) / span * (height - 20)   # invert: higher temp -> higher on screen
        points.append(f"{x:.1f},{y:.1f}")
    path = " ".join(points)
    return (
        f'<svg viewBox="0 0 {width} {height}" width="100%" style="max-width:{width}px">'
        f'<polyline fill="none" stroke="#305496" stroke-width="2" points="{path}"/>'
        f'<text x="10" y="14" font-size="11" fill="#666">{hi:.1f}°C</text>'
        f'<text x="10" y="{height-2}" font-size="11" fill="#666">{lo:.1f}°C</text>'
        f"</svg>"
    )


def render(history: list[dict]) -> None:
    latest = history[-1]
    temps = [h["temp"] for h in history][-48:]    # last 48 readings
    chart = sparkline_svg(temps)
    rows = "".join(
        f"<tr><td>{h['time']}</td><td>{h['temp']:.1f}°C</td></tr>"
        for h in reversed(history[-12:])
    )
    html = f"""<!doctype html>
<html><head><meta charset="utf-8"><title>Weather Dashboard</title>
<style>
  body {{ font-family: system-ui, sans-serif; max-width: 700px; margin: 2rem auto; color:#222; }}
  .big {{ font-size: 3rem; font-weight: 700; }}
  table {{ border-collapse: collapse; width: 100%; margin-top: 1rem; }}
  td, th {{ border-bottom: 1px solid #ddd; padding: 6px 10px; text-align: left; }}
</style></head><body>
  <h1>{CITY} - current temperature</h1>
  <div class="big">{latest['temp']:.1f}°C</div>
  <p>Updated {latest['time']}</p>
  <h2>Recent trend</h2>
  {chart}
  <h2>Last 12 readings</h2>
  <table><tr><th>Time</th><th>Temp</th></tr>{rows}</table>
</body></html>"""
    DASHBOARD.write_text(html, encoding="utf-8")


def main() -> None:
    history = load_history()
    temp = fetch_temp()
    history.append({"time": datetime.now().strftime("%Y-%m-%d %H:%M"), "temp": temp})
    save_history(history)
    render(history)
    print(f"{CITY}: {temp:.1f}°C  ->  updated {DASHBOARD} ({len(history)} readings total)")


if __name__ == "__main__":
    main()
```

### How to run, then schedule
```bash
pip install requests
python weather_dashboard.py          # run once; open dashboard.html in a browser
```

Schedule it every hour with cron (macOS/Linux). `crontab -e`, then add:

```cron
# minute hour day month weekday   command
0 * * * * cd /path/to/project && /usr/bin/python3 weather_dashboard.py >> run.log 2>&1
```

That means "at minute 0 of every hour." The `>> run.log 2>&1` captures stdout *and* stderr so you can debug failures. On Windows, use Task Scheduler instead. Each run appends one reading and rebuilds the dashboard, so the chart fills in over time, a genuine self-updating pipeline.

## 🧪 Project 5 - PDF toolkit (merge / split / extract text)

> Goal: A single CLI that merges PDFs, splits a PDF into pages, and extracts text, the three operations you always end up needing. Built on `pypdf` (pure Python, `pip install pypdf`).

### Design
A subcommand CLI (`argparse` subparsers): `merge`, `split`, `text`. Each is a small, focused function reading/writing PDF objects.

### Full code

```python
#!/usr/bin/env python3
"""pdftool.py - merge, split, or extract text from PDFs.

Usage:
    python pdftool.py merge out.pdf a.pdf b.pdf c.pdf
    python pdftool.py split big.pdf --outdir pages/
    python pdftool.py text  doc.pdf            # prints extracted text
    python pdftool.py text  doc.pdf --pages 1-3
"""
from __future__ import annotations

import argparse
from pathlib import Path

from pypdf import PdfReader, PdfWriter


def merge(output: Path, inputs: list[Path]) -> None:
    writer = PdfWriter()
    for pdf in inputs:
        reader = PdfReader(str(pdf))
        for page in reader.pages:
            writer.add_page(page)
        print(f"  + {pdf.name} ({len(reader.pages)} pages)")
    with output.open("wb") as f:
        writer.write(f)
    print(f"Merged {len(inputs)} files -> {output}")


def split(source: Path, outdir: Path) -> None:
    outdir.mkdir(parents=True, exist_ok=True)
    reader = PdfReader(str(source))
    for i, page in enumerate(reader.pages, start=1):
        writer = PdfWriter()
        writer.add_page(page)
        dest = outdir / f"{source.stem}_page_{i:03d}.pdf"
        with dest.open("wb") as f:
            writer.write(f)
        print(f"  wrote {dest.name}")
    print(f"Split {len(reader.pages)} pages -> {outdir}/")


def parse_page_range(spec: str | None, total: int) -> range:
    """'1-3' -> pages 1,2,3 (1-indexed, inclusive). None -> all pages."""
    if not spec:
        return range(1, total + 1)
    if "-" in spec:
        a, b = spec.split("-", 1)
        return range(int(a), int(b) + 1)
    p = int(spec)
    return range(p, p + 1)


def extract_text(source: Path, pages_spec: str | None) -> None:
    reader = PdfReader(str(source))
    total = len(reader.pages)
    for pno in parse_page_range(pages_spec, total):
        if 1 <= pno <= total:
            text = reader.pages[pno - 1].extract_text() or ""
            print(f"\n----- Page {pno} -----")
            print(text.strip())


def main() -> None:
    ap = argparse.ArgumentParser(description="Tiny PDF toolkit.")
    sub = ap.add_subparsers(dest="cmd", required=True)

    m = sub.add_parser("merge", help="merge PDFs into one")
    m.add_argument("output", type=Path)
    m.add_argument("inputs", type=Path, nargs="+")

    s = sub.add_parser("split", help="split a PDF into one-file-per-page")
    s.add_argument("source", type=Path)
    s.add_argument("--outdir", type=Path, default=Path("pages"))

    t = sub.add_parser("text", help="extract text")
    t.add_argument("source", type=Path)
    t.add_argument("--pages", help="e.g. '1-3' or '2' (default: all)")

    args = ap.parse_args()
    if args.cmd == "merge":
        merge(args.output, args.inputs)
    elif args.cmd == "split":
        split(args.source, args.outdir)
    elif args.cmd == "text":
        extract_text(args.source, args.pages)


if __name__ == "__main__":
    main()
```

## 🧪 Project 6 - "Automate something annoying": a bulk file renamer

> Goal: The single most common annoying task, renaming a pile of files with a consistent scheme. Add a sequential number, a prefix, a date stamp, or find-and-replace text in names. Dry-run by default, collision-safe. This is the everyday automation everyone wishes they had.

```python
#!/usr/bin/env python3
"""rename.py - bulk-rename files with a numbering/prefix/replace scheme.

Usage:
    python rename.py ~/scans --prefix invoice --number          # invoice_001.pdf, ...
    python rename.py ~/pics --replace " " _                     # spaces -> underscores
    python rename.py ~/pics --prefix trip --number --apply      # actually rename
"""
from __future__ import annotations

import argparse
from pathlib import Path


def planned_name(item: Path, idx: int, args) -> str:
    stem = item.stem
    if args.replace is not None:
        stem = stem.replace(args.replace[0], args.replace[1])
    if args.prefix:
        stem = f"{args.prefix}_{stem}" if not args.number else args.prefix
    if args.number:
        stem = f"{stem}_{idx:0{args.pad}d}" if args.prefix else f"{idx:0{args.pad}d}"
    return stem + item.suffix.lower()


def unique(dest: Path) -> Path:
    n = 1
    cand = dest
    while cand.exists():
        cand = dest.with_stem(f"{dest.stem}_dup{n}")
        n += 1
    return cand


def main() -> None:
    ap = argparse.ArgumentParser(description="Bulk file renamer.")
    ap.add_argument("folder", type=Path)
    ap.add_argument("--prefix", help="prefix for each file")
    ap.add_argument("--number", action="store_true", help="append a sequence number")
    ap.add_argument("--pad", type=int, default=3, help="zero-padding width for numbers")
    ap.add_argument("--replace", nargs=2, metavar=("OLD", "NEW"),
                    help="replace text in filenames, e.g. --replace ' ' _")
    ap.add_argument("--ext", help="only rename files with this extension, e.g. .jpg")
    ap.add_argument("--apply", action="store_true", help="actually rename (default: dry run)")
    args = ap.parse_args()

    files = sorted(p for p in args.folder.expanduser().iterdir() if p.is_file())
    if args.ext:
        files = [p for p in files if p.suffix.lower() == args.ext.lower()]

    if not files:
        print("No matching files.")
        return

    renamed = 0
    for idx, item in enumerate(files, start=1):
        new_name = planned_name(item, idx, args)
        dest = unique(item.with_name(new_name))
        if dest.name == item.name:
            continue
        if args.apply:
            item.rename(dest)
            print(f"RENAMED {item.name}  ->  {dest.name}")
        else:
            print(f"PLAN    {item.name}  ->  {dest.name}")
        renamed += 1

    verb = "Renamed" if args.apply else "Would rename"
    print(f"\n{verb} {renamed} file(s)." + ("" if args.apply else " Add --apply to do it."))


if __name__ == "__main__":
    main()
```

```bash

python rename.py ~/scans --prefix invoice --number --apply  # do it
python rename.py ~/photos --replace " " _ --ext .jpg --apply
```

## 🧪 Project 7 - "Automate something annoying": a folder watcher that auto-sorts new files

> Goal: Instead of organizing a folder on demand, *watch* it and react the instant a file lands, auto-filing downloads as they arrive. This introduces the **polling loop**, the simplest form of event-driven automation, and is genuinely delightful to leave running.

The idea: every few seconds, list the folder, compare against what we saw last time, and handle anything new. No extra dependencies (a more robust version would use the `watchdog` package, but polling teaches the concept and works everywhere).

```python
#!/usr/bin/env python3
"""watcher.py - watch a folder and auto-file new files by extension.

Usage:
    python watcher.py ~/Downloads            # watch + sort (Ctrl-C to stop)
    python watcher.py ~/Downloads --interval 5
"""
from __future__ import annotations

import argparse
import shutil
import time
from pathlib import Path

CATEGORIES = {
    "Images": {".jpg", ".jpeg", ".png", ".gif", ".webp", ".heic"},
    "Documents": {".pdf", ".docx", ".txt", ".md", ".rtf"},
    "Spreadsheets": {".csv", ".xlsx", ".xls"},
    "Archives": {".zip", ".tar", ".gz", ".rar", ".7z"},
    "Installers": {".dmg", ".pkg", ".exe", ".msi", ".deb"},
}


def category_for(path: Path) -> str:
    suffix = path.suffix.lower()
    for name, exts in CATEGORIES.items():
        if suffix in exts:
            return name
    return "Other"


def unique(dest: Path) -> Path:
    n = 1
    cand = dest
    while cand.exists():
        cand = dest.with_stem(f"{dest.stem} ({n})")
        n += 1
    return cand


def file_it(item: Path, root: Path) -> None:
    dest_dir = root / category_for(item)
    dest_dir.mkdir(exist_ok=True)
    dest = unique(dest_dir / item.name)
    shutil.move(str(item), str(dest))
    print(f"[{time.strftime('%H:%M:%S')}] filed {item.name} -> {dest.relative_to(root)}")


def watch(root: Path, interval: float) -> None:
    print(f"Watching {root} every {interval}s. Ctrl-C to stop.")
    seen: set[str] = {p.name for p in root.iterdir() if p.is_file()}
    try:
        while True:
            time.sleep(interval)
            for item in root.iterdir():
                # only handle settled files we haven't seen (skip our own category dirs)
                if item.is_file() and item.name not in seen and not item.name.startswith("."):
                    # crude "is it done downloading?" check: size stable across a beat
                    size1 = item.stat().st_size
                    time.sleep(0.5)
                    if item.exists() and item.stat().st_size == size1:
                        file_it(item, root)
            seen = {p.name for p in root.iterdir() if p.is_file()}
    except KeyboardInterrupt:
        print("\nStopped.")


def main() -> None:
    ap = argparse.ArgumentParser(description="Watch a folder and auto-file new files.")
    ap.add_argument("folder", type=Path)
    ap.add_argument("--interval", type=float, default=3.0, help="seconds between scans")
    args = ap.parse_args()
    watch(args.folder.expanduser(), args.interval)


if __name__ == "__main__":
    main()
```

### How to run
```bash
python watcher.py ~/Downloads --interval 3
# drop a file into ~/Downloads and watch it get filed automatically
```

### What to notice
- The **polling loop** (`while True: sleep; check for new`) is the simplest event loop there is. Real frameworks fire OS file-system events instead of polling, but the mental model is identical.
- The "size stable across a beat" check is a pragmatic way to avoid grabbing a file *mid-download*, a real gotcha in folder automation.
- `KeyboardInterrupt` gives a clean Ctrl-C exit instead of an ugly traceback.

### Extend it
- Swap polling for the `watchdog` package for instant, CPU-friendly reactions.
- Add per-extension rules from a JSON config.
- Log every move to a file so you can audit (or undo) what the watcher did.

## 🧪 Project 8 - "Automate something annoying": a backup snapshotter

> Goal: One command that zips a folder into a timestamped archive, drops it in a backups directory, and prunes old snapshots so you keep, say, the last 7. The boring chore everyone *means* to do and never does, so automate it and schedule it.

```python
#!/usr/bin/env python3
"""backup.py - make a timestamped zip snapshot of a folder, keep the last N.

Usage:
    python backup.py ~/Projects/important ~/Backups
    python backup.py ~/Projects/important ~/Backups --keep 10
"""
from __future__ import annotations

import argparse
import shutil
from datetime import datetime
from pathlib import Path


def make_snapshot(source: Path, dest_dir: Path) -> Path:
    dest_dir.mkdir(parents=True, exist_ok=True)
    stamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    base = dest_dir / f"{source.name}_{stamp}"
    # make_archive adds the .zip and zips the whole tree for us
    archive = shutil.make_archive(str(base), "zip", root_dir=source)
    return Path(archive)


def prune(dest_dir: Path, source_name: str, keep: int) -> list[Path]:
    snaps = sorted(dest_dir.glob(f"{source_name}_*.zip"))  # names sort chronologically
    to_delete = snaps[:-keep] if keep > 0 else []
    for old in to_delete:
        old.unlink()
    return to_delete


def main() -> None:
    ap = argparse.ArgumentParser(description="Timestamped folder backups with retention.")
    ap.add_argument("source", type=Path, help="folder to back up")
    ap.add_argument("dest", type=Path, help="where to store snapshots")
    ap.add_argument("--keep", type=int, default=7, help="how many snapshots to retain")
    args = ap.parse_args()

    source = args.source.expanduser()
    if not source.is_dir():
        raise SystemExit(f"Not a directory: {source}")

    archive = make_snapshot(source, args.dest.expanduser())
    size_mb = archive.stat().st_size / 1_048_576
    print(f"Created {archive.name} ({size_mb:.1f} MB)")

    deleted = prune(args.dest.expanduser(), source.name, args.keep)
    for d in deleted:
        print(f"Pruned old snapshot: {d.name}")
    print(f"Keeping the most recent {args.keep} snapshot(s).")


if __name__ == "__main__":
    main()
```

### How to run, then schedule
```bash
python backup.py ~/Projects/important ~/Backups --keep 7
```

Daily backup at 2am via cron (`crontab -e`):
```cron
0 2 * * * /usr/bin/python3 /path/to/backup.py ~/Projects/important ~/Backups --keep 7 >> ~/Backups/backup.log 2>&1
```

### What to notice
- `shutil.make_archive` zips an entire tree in one call, no manual file walking.
- **Retention by sorting timestamped names**: because the timestamp format sorts chronologically as text, `sorted(...)[:-keep]` is "everything except the newest N." Clean and dependency-free.
- This is **read → transform → write → schedule** with a *retention* twist; pruning old output is a real concern in any pipeline that accumulates files.

### Extend it
- Add `--incremental` to only snapshot if files changed since the last backup (compare hashes from Chapter 4.2).
- Encrypt the zip, or push it to cloud storage (S3) after creating it.
- Email a success/failure summary (Project 2) so you know the cron job is alive.

## ⚠️ Common mistakes

- **No dry-run on a destructive tool.** Moving/renaming/deleting at scale with a bug ruins *everything at once*. Default to printing a plan; require `--apply` to act.
- **Overwriting the source.** Always write transformed output to a *new* file (`report.xlsx`, not back into `sales.xlsx`). Use `shutil.copy2` to preserve metadata before transforming.
- **Filename collisions silently clobbering data.** Two files mapping to the same destination = one gets destroyed. Use a `unique()` helper that appends ` (1)`, `_dup1`, etc.
- **Hard-coding secrets (SMTP passwords, API keys).** Use env vars; for Gmail use an *app password*, never your real one (Chapter 4.2).
- **Sending real emails while testing.** Always `--dry-run` the mailer first. A loop bug that emails everyone twice is mortifying and sometimes unrecoverable.
- **Assuming EXIF/text always exists.** Photos may lack EXIF; scanned PDFs have no text layer. Always have a fallback (file mtime; OCR) and check for empty results.
- **Reading formulas instead of values from Excel.** `load_workbook(path, data_only=True)` returns computed values; without it you get the formula strings.
- **Cron job runs but does nothing / can't find files.** Cron starts in a minimal environment with a bare PATH and a different working directory. Use *absolute paths* for both Python and your script, `cd` into the project, and redirect output to a log (`>> run.log 2>&1`) so you can debug.
- **A watcher grabbing a half-downloaded file.** Check the size is stable (or wait for the temp/partial extension to disappear) before acting.
- **No logging.** When an unattended automation misbehaves at 3am, a log is the only evidence. Print/append timestamps and what happened.

```python
#!/usr/bin/env python3
"""tidy.py - organize | dedupe | backup, dry-run by default."""
import argparse, hashlib, shutil
from collections import defaultdict
from datetime import datetime
from pathlib import Path

def unique(dest: Path) -> Path:
    n = 1; cand = dest
    while cand.exists():
        cand = dest.with_stem(f"{dest.stem} ({n})"); n += 1
    return cand

def cmd_organize(args):
    cats = {".jpg": "Images", ".png": "Images", ".pdf": "Documents", ".csv": "Spreadsheets"}
    for p in args.folder.iterdir():
        if not p.is_file():
            continue
        dest = unique(args.folder / cats.get(p.suffix.lower(), "Other") / p.name)
        print(("MOVE " if args.apply else "PLAN ") + f"{p.name} -> {dest.relative_to(args.folder)}")
        if args.apply:
            dest.parent.mkdir(exist_ok=True); shutil.move(str(p), str(dest))

def cmd_dedupe(args):
    by_hash = defaultdict(list)
    for p in args.folder.rglob("*"):
        if p.is_file():
            by_hash[hashlib.sha256(p.read_bytes()).hexdigest()].append(p)
    for paths in by_hash.values():
        for extra in sorted(paths)[1:]:
            print(("DEL  " if args.apply else "DUP  ") + str(extra))
            if args.apply:
                extra.unlink()

def cmd_backup(args):
    stamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    if args.apply:
        out = shutil.make_archive(str(args.folder.parent / f"{args.folder.name}_{stamp}"),
                                  "zip", root_dir=args.folder)
        print(f"created {out}")
    else:
        print(f"PLAN would create {args.folder.name}_{stamp}.zip")

ap = argparse.ArgumentParser(description="tidy: organize | dedupe | backup")
ap.add_argument("--apply", action="store_true")
sub = ap.add_subparsers(dest="cmd", required=True)
for name in ("organize", "dedupe", "backup"):
    sp = sub.add_parser(name); sp.add_argument("folder", type=Path)
args = ap.parse_args(); args.folder = args.folder.expanduser()
```
